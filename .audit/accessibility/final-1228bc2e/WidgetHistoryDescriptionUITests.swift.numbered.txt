1: import XCTest
2: 
3: final class WidgetHistoryDescriptionUITests: XCTestCase {
4:     @MainActor
5:     func testOwnDescriptionCanBeCorrectedReopenedCancelledAndClearedInHistory() {
6:         continueAfterFailure = false
7:         let app = XCUIApplication()
8:         app.launchArguments = ["-paeonia-scenario", "widget-history-populated",
9:                                "-paeonia-scenario-hide-controls", "YES",
10:                                "-AppleLanguages", "(en)", "-AppleLocale", "en_US"]
11:         app.launch()
12:         defer { app.terminate() }
13:         let edit = app.buttons["media.alternative.edit"].firstMatch
14:         scrollTo(edit, app: app)
15:         edit.tap()
16:         let field = app.descendants(matching: .any)["media.alternative.text"].firstMatch
17:         XCTAssertTrue(field.waitForExistence(timeout: 5))
18:         field.tap()
19:         field.typeText("A sun above two hills")
20:         app.buttons["media.alternative.save"].tap()
21:         XCTAssertTrue(app.staticTexts["A sun above two hills"].waitForExistence(timeout: 5))
22:         scrollTo(edit, app: app)
23:         edit.tap()
24:         XCTAssertEqual(field.value as? String, "A sun above two hills")
25:         app.buttons["media.alternative.cancel"].tap()
26:         scrollTo(edit, app: app)
27:         edit.tap()
28:         let clear = app.buttons["media.alternative.clear"]
29:         scrollTo(clear, app: app)
30:         clear.tap()
31:         app.buttons["media.alternative.save"].tap()
32:         scrollTo(edit, app: app)
33:         edit.tap()
34:         XCTAssertTrue(field.waitForExistence(timeout: 5))
35:         // Reopening resets the Form scroll position. Find the row before reading its state.
36:         scrollTo(clear, app: app)
37:         XCTAssertFalse(clear.isEnabled)
38:         app.buttons["media.alternative.cancel"].tap()
39:     }
40: 
41:     @MainActor
42:     private func scrollTo(_ element: XCUIElement, app: XCUIApplication) {
43:         for _ in 0..<20 where !element.exists || !element.isHittable {
44:             if element.exists && element.frame.midY < app.windows.firstMatch.frame.midY {
45:                 app.swipeDown()
46:             } else {
47:                 app.swipeUp()
48:             }
49:         }
50:         XCTAssertTrue(element.isHittable)
51:     }
52: }
